    76→/* ============ IMAGE UPLOADER COMPONENT ============ */
    77→function ImageUploader({ onUpload, label }: { onUpload: (url: string) => void; label: string }) {
    78→  const inputRef = useRef<HTMLInputElement>(null);
    79→  const [uploading, setUploading] = useState(false);
    80→
    81→  const handleFile = async (file: File) => {
    82→    setUploading(true);
    83→    try {
    84→      const formData = new FormData();
    85→      formData.append("file", file);
    86→      const r = await fetch("/api/upload", { method: "POST", body: formData });
    87→      const j = await r.json();
    88→      if (j.error) {
    89→        toast.error(j.error);
    90→      } else {
    91→        onUpload(j.url);
    92→        toast.success("Image uploaded");
    93→      }
    94→    } catch {
    95→      toast.error("Upload failed");
    96→    }
    97→    setUploading(false);
    98→  };
    99→
   100→  return (
   101→    <div>
   102→      <input
   103→        ref={inputRef}
   104→        type="file"
   105→        accept="image/jpeg,image/png,image/webp"
   106→        className="hidden"
   107→        onChange={(e) => {
   108→          const file = e.target.files?.[0];
   109→          if (file) handleFile(file);
   110→        }}
   111→      />
   112→      <button
   113→        onClick={() => inputRef.current?.click()}
   114→        disabled={uploading}
   115→        className="inline-flex items-center gap-2 rounded-lg border border-champagne/20 bg-ink/50 px-3 py-2 text-xs font-semibold text-champagne transition-colors hover:bg-champagne/10 disabled:opacity-40"
   116→      >
   117→        {uploading ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Upload className="h-3.5 w-3.5" />}
   118→        {uploading ? "Uploading…" : label}
   119→      </button>
   120→    </div>
   121→  );
   122→}

   210→function EditableField({ label, value, type = "text", onSave }: { label: string; value: string; type?: string; onSave: (v: string) => void }) {
   211→  const [editing, setEditing] = useState(false);
   212→  const [val, setVal] = useState(value);
   213→
   214→  return (
   215→    <div>
   216→      <label className="text-[10px] uppercase tracking-wider text-ivory/50">{label}</label>
   217→      {editing ? (
   218→        <div className="flex gap-1">
   219→          <input type={type} value={val} onChange={(e) => setVal(e.target.value)} className="flex-1 rounded-lg border border-champagne/15 bg-ink px-2 py-1 text-sm text-ivory focus:border-champagne/40 focus:outline-none" />
   220→          <button onClick={() => { onSave(val); setEditing(false); toast.success(`${label} updated`); }} className="rounded bg-green-500/20 px-2 text-xs text-green-300">
   221→            <Check className="h-3 w-3" />
   222→          </button>
   223→          <button onClick={() => { setVal(value); setEditing(false); }} className="rounded bg-red-500/20 px-2 text-xs text-red-300">
   224→            <X className="h-3 w-3" />
   225→          </button>
   226→        </div>
   227→      ) : (
   228→        <button onClick={() => setEditing(true)} className="block w-full truncate text-left text-sm text-ivory hover:text-champagne">
   229→          {value} <span className="text-[10px] text-ivory/30">✎</span>
   230→        </button>
   231→      )}
   232→    </div>
   233→  );
   234→}

   504→/* ============ GENERIC CMS (for features, events, testimonials, faqs, trustBadges) ============ */
   505→interface CMSField {
   506→  key: string;
   507→  label: string;
   508→  textarea?: boolean;
   509→  image?: boolean;
   510→}
   511→
   512→function GenericCMS({ type, title, fields }: { type: string; title: string; fields: CMSField[] }) {
   513→  const [items, setItems] = useState<any[]>([]);
   514→  const [showAdd, setShowAdd] = useState(false);
   515→  const [newItem, setNewItem] = useState<Record<string, string>>({});
   516→
   517→  const load = () => {
   518→    fetch(`/api/cms?type=${type}`, { cache: "no-store" })
   519→      .then(r => r.json())
   520→      .then(j => setItems(j.data || []));
   521→  };
   522→  useEffect(() => { load(); }, [type]);
   523→
   524→  const addItem = async () => {
   525→    const data: any = { ...newItem, sortOrder: items.length };
   526→    if (newItem.rating) data.rating = parseInt(newItem.rating);
   527→    await fetch("/api/cms", {
   528→      method: "POST",
   529→      headers: { "Content-Type": "application/json" },
   530→      body: JSON.stringify({ type, data }),
   531→    });
   532→    toast.success(`${title} item added`);
   533→    setShowAdd(false);
   534→    setNewItem({});
   535→    load();
   536→  };
   537→
   538→  const deleteItem = async (id: string) => {
   539→    if (!confirm("Delete this item?")) return;
   540→    await fetch(`/api/cms?type=${type}&id=${id}`, { method: "DELETE" });
   541→    toast.success("Deleted");
   542→    load();
   543→  };
   544→
   545→  const updateField = async (id: string, field: string, value: any) => {
   546→    const data: any = { [field]: field === "rating" ? parseInt(value) : value };
   547→    await fetch("/api/cms", {
   548→      method: "PATCH",
   549→      headers: { "Content-Type": "application/json" },
   550→      body: JSON.stringify({ type, id, data }),
   551→    });
   552→    toast.success(`${field} updated`);
   553→    load();
   554→  };
   555→
   556→  return (
   557→    <div>
   558→      <div className="flex items-center justify-between">
   559→        <div>
   560→          <h2 className="font-serif text-2xl text-ivory">{title}</h2>
   561→          <p className="mt-1 text-sm text-ivory/60">Add, edit, or delete {title.toLowerCase()}. Changes appear on the website instantly.</p>
   562→        </div>
   563→        <button onClick={() => setShowAdd(!showAdd)} className="btn-luxe text-xs">
   564→          <Plus className="h-4 w-4" /> Add
   565→        </button>
   566→      </div>
   567→
   568→      {/* Add form */}
   569→      {showAdd && (
   570→        <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} className="mt-4 overflow-hidden">
   571→          <div className="card-luxe p-5">
   572→            <div className="grid gap-3 sm:grid-cols-2">
   573→              {fields.map((f) => (
   574→                <div key={f.key} className={f.textarea ? "sm:col-span-2" : ""}>
   575→                  <label className="text-[10px] uppercase tracking-wider text-ivory/50">{f.label}</label>
   576→                  {f.textarea ? (
   577→                    <textarea
   578→                      value={newItem[f.key] || ""}
   579→                      onChange={(e) => setNewItem({ ...newItem, [f.key]: e.target.value })}
   580→                      rows={3}
   581→                      className="mt-1 w-full rounded-lg border border-champagne/15 bg-ink px-3 py-2 text-sm text-ivory focus:border-champagne/40 focus:outline-none resize-none"
   582→                      placeholder={f.label}
   583→                    />
   584→                  ) : f.image ? (
   585→                    <div className="mt-1 flex items-center gap-2">
   586→                      <input
   587→                        value={newItem[f.key] || ""}
   588→                        onChange={(e) => setNewItem({ ...newItem, [f.key]: e.target.value })}
   589→                        className="flex-1 rounded-lg border border-champagne/15 bg-ink px-3 py-2 text-sm text-ivory focus:outline-none"
   590→                        placeholder="Image URL or upload"
   591→                      />
   592→                      <ImageUploader label="Upload" onUpload={(url) => setNewItem({ ...newItem, [f.key]: url })} />
   593→                    </div>
   594→                  ) : (
   595→                    <input
   596→                      value={newItem[f.key] || ""}
   597→                      onChange={(e) => setNewItem({ ...newItem, [f.key]: e.target.value })}
   598→                      className="mt-1 w-full rounded-lg border border-champagne/15 bg-ink px-3 py-2 text-sm text-ivory focus:border-champagne/40 focus:outline-none"
   599→                      placeholder={f.label}
   600→                    />
   601→                  )}
   602→                </div>
   603→              ))}
   604→            </div>
   605→            <div className="mt-4 flex gap-2">
   606→              <button onClick={addItem} className="btn-luxe text-xs">Add Item</button>
   607→              <button onClick={() => setShowAdd(false)} className="btn-ghost-luxe text-xs">Cancel</button>
   608→            </div>
   609→          </div>
   610→        </motion.div>
   611→      )}
   612→
   613→      {/* Items list */}
   614→      <div className="mt-6 space-y-3">
   615→        {items.map((item, idx) => (
   616→          <div key={item.id} className="card-luxe p-4">
   617→            <div className="flex items-start justify-between gap-4">
   618→              <div className="flex-1 min-w-0">
   619→                {/* Show image if exists */}
   620→                {item.image && (
   621→                  <img src={item.image} alt="" className="mb-2 h-16 w-24 rounded-lg object-cover photo-cinematic" />
   622→                )}
   623→                {/* Inline editable fields */}
   624→                <div className="grid gap-2 sm:grid-cols-2">
   625→                  {fields.map((f) => (
   626→                    <EditableField
   627→                      key={f.key}
   628→                      label={f.label}
   629→                      value={String(item[f.key] || "")}
   630→                      onSave={(v) => updateField(item.id, f.key, v)}
   631→                    />
   632→                  ))}
   633→                </div>
   634→                {/* Image upload for image fields */}
   635→                {fields.some(f => f.image) && (
   636→                  <div className="mt-2">
   637→                    <ImageUploader label="Change Image" onUpload={(url) => updateField(item.id, "image", url)} />
   638→                  </div>
   639→                )}
   640→              </div>
   641→              <button
   642→                onClick={() => deleteItem(item.id)}
   643→                className="rounded-full bg-red-500/10 px-2 py-1 text-[10px] font-semibold text-red-300 hover:bg-red-500/20"
   644→              >
   645→                <Trash2 className="h-3 w-3" />
   646→              </button>
   647→            </div>
   648→          </div>
   649→        ))}
   650→        {items.length === 0 && (
   651→          <p className="py-8 text-center text-sm text-ivory/50">No {title.toLowerCase()} yet. Click "Add" to create one.</p>
   652→        )}
   653→      </div>
   654→    </div>
   655→  );
   656→}
