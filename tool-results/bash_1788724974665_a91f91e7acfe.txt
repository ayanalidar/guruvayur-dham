    81→/* ============ IMAGE UPLOADER COMPONENT ============ */
    82→function ImageUploader({ onUpload, label }: { onUpload: (url: string) => void; label: string }) {
    83→  const inputRef = useRef<HTMLInputElement>(null);
    84→  const [uploading, setUploading] = useState(false);
    85→
    86→  const handleFile = async (file: File) => {
    87→    setUploading(true);
    88→    try {
    89→      const formData = new FormData();
    90→      formData.append("file", file);
    91→      const r = await fetch("/api/upload", { method: "POST", body: formData });
    92→      const j = await r.json();
    93→      if (j.error) {
    94→        toast.error(j.error);
    95→      } else {
    96→        onUpload(j.url);
    97→        toast.success("Image uploaded");
    98→      }
    99→    } catch {
   100→      toast.error("Upload failed");
   101→    }
   102→    setUploading(false);
   103→  };
   104→
   105→  return (
   106→    <div>
   107→      <input
   108→        ref={inputRef}
   109→        type="file"
   110→        accept="image/jpeg,image/png,image/webp"
   111→        className="hidden"
   112→        onChange={(e) => {
   113→          const file = e.target.files?.[0];
   114→          if (file) handleFile(file);
   115→        }}
   116→      />
   117→      <button
   118→        onClick={() => inputRef.current?.click()}
   119→        disabled={uploading}
   120→        className="inline-flex items-center gap-2 rounded-lg border border-champagne/20 bg-ink/50 px-3 py-2 text-xs font-semibold text-champagne transition-colors hover:bg-champagne/10 disabled:opacity-40"
   121→      >
   122→        {uploading ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Upload className="h-3.5 w-3.5" />}
   123→        {uploading ? "Uploading…" : label}
   124→      </button>
   125→    </div>
   126→  );
   127→}

   378→function EditableField({ label, value, type = "text", onSave }: { label: string; value: string; type?: string; onSave: (v: string) => void }) {
   379→  const [editing, setEditing] = useState(false);
   380→  const [val, setVal] = useState(value);
   381→
   382→  return (
   383→    <div>
   384→      <label className="text-[10px] uppercase tracking-wider text-ivory/50">{label}</label>
   385→      {editing ? (
   386→        <div className="flex gap-1">
   387→          <input type={type} value={val} onChange={(e) => setVal(e.target.value)} className="flex-1 rounded-lg border border-champagne/15 bg-ink px-2 py-1 text-sm text-ivory focus:border-champagne/40 focus:outline-none" />
   388→          <button onClick={() => { onSave(val); setEditing(false); toast.success(`${label} updated`); }} className="rounded bg-green-500/20 px-2 text-xs text-green-300">
   389→            <Check className="h-3 w-3" />
   390→          </button>
   391→          <button onClick={() => { setVal(value); setEditing(false); }} className="rounded bg-red-500/20 px-2 text-xs text-red-300">
   392→            <X className="h-3 w-3" />
   393→          </button>
   394→        </div>
   395→      ) : (
   396→        <button onClick={() => setEditing(true)} className="block w-full truncate text-left text-sm text-ivory hover:text-champagne">
   397→          {value} <span className="text-[10px] text-ivory/30">✎</span>
   398→        </button>
   399→      )}
   400→    </div>
   401→  );
   402→}

   672→/* ============ GENERIC CMS (for features, events, testimonials, faqs, trustBadges) ============ */
   673→interface CMSField {
   674→  key: string;
   675→  label: string;
   676→  textarea?: boolean;
   677→  image?: boolean;

   680→function GenericCMS({ type, title, fields }: { type: string; title: string; fields: CMSField[] }) {
   681→  const [items, setItems] = useState<any[]>([]);
   682→  const [showAdd, setShowAdd] = useState(false);
   683→  const [newItem, setNewItem] = useState<Record<string, string>>({});
   684→
   685→  const load = () => {
   686→    fetch(`/api/cms?type=${type}`, { cache: "no-store" })
   687→      .then(r => r.json())
   688→      .then(j => setItems(j.data || []));
   689→  };
   690→  useEffect(() => { load(); }, [type]);
   691→
   692→  const addItem = async () => {
   693→    const data: any = { ...newItem, sortOrder: items.length };
   694→    if (newItem.rating) data.rating = parseInt(newItem.rating);
   695→    await fetch("/api/cms", {
   696→      method: "POST",
   697→      headers: { "Content-Type": "application/json" },
   698→      body: JSON.stringify({ type, data }),
   699→    });
   700→    toast.success(`${title} item added`);
   701→    setShowAdd(false);
   702→    setNewItem({});
   703→    load();
   704→  };
   705→
   706→  const deleteItem = async (id: string) => {
   707→    if (!confirm("Delete this item?")) return;
   708→    await fetch(`/api/cms?type=${type}&id=${id}`, { method: "DELETE" });

   710→    load();
   711→  };
   712→
   713→  const updateField = async (id: string, field: string, value: any) => {

   715→    await fetch("/api/cms", {
   716→      method: "PATCH",
   717→      headers: { "Content-Type": "application/json" },
   718→      body: JSON.stringify({ type, id, data }),

   743→                  <label className="text-[10px] uppercase tracking-wider text-ivory/50">{f.label}</label>
   744→                  {f.textarea ? (
   745→                    <textarea
   746→                      value={newItem[f.key] || ""}
   747→                      onChange={(e) => setNewItem({ ...newItem, [f.key]: e.target.value })}
   748→                      rows={3}
   749→                      className="mt-1 w-full rounded-lg border border-champagne/15 bg-ink px-3 py-2 text-sm text-ivory focus:border-champagne/40 focus:outline-none resize-none"
   750→                      placeholder={f.label}

   758→                        placeholder="Image URL or upload"

   760→                      <ImageUploader label="Upload" onUpload={(url) => setNewItem({ ...newItem, [f.key]: url })} />
