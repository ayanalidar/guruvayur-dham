    17→  name          String

    19→  price         Int

    22→  reviews       Int          @default(0)
    23→  capacity      Int
    24→  size          String
    25→  bedType       String
    26→  image         String
    27→  gallery       String       // JSON array of URLs
    28→  badge         String?

    30→  shortDesc     String
    31→  amenities     String       // JSON array of strings
    32→  totalUnits    Int          @default(1) // how many physical rooms of this type
    33→  active        Boolean      @default(true)
    34→  createdAt     DateTime     @default(now())
    35→  updatedAt     DateTime     @updatedAt
    36→  bookings      Booking[]
    37→  rates         RatePlan[]
    38→  availability  Availability[]
    39→}

   120→  id              String         @id @default(cuid())

   123→  date            DateTime       // the specific night

   128→  createdAt       DateTime       @default(now())
   129→  updatedAt       DateTime       @updatedAt
   130→
   131→  @@unique([roomId, date])
