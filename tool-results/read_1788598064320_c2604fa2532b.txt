    76→    76→/* ============ IMAGE UPLOADER COMPONENT ============ */
    77→    77→function ImageUploader({ onUpload, label }: { onUpload: (url: string) => void; label: string }) {
    78→    78→  const inputRef = useRef<HTMLInputElement>(null);
    79→    79→  const [uploading, setUploading] = useState(false);
    80→    80→
    81→    81→  const handleFile = async (file: File) => {
    82→    82→    setUploading(true);
    83→    83→    try {
    84→    84→      const formData = new FormData();
    85→    85→      formData.append("file", file);
    86→    86→      const r = await fetch("/api/upload", { method: "POST", body: formData });
    87→    87→      const j = await r.json();
    88→    88→      if (j.error) {
    89→    89→        toast.error(j.error);
    90→    90→      } else {
    91→    91→        onUpload(j.url);
    92→    92→        toast.success("Image uploaded");
    93→    93→      }
    94→    94→    } catch {
    95→    95→      toast.error("Upload failed");
    96→    96→    }
    97→    97→    setUploading(false);
    98→    98→  };
    99→    99→
   100→   100→  return (
   101→   101→    <div>
   102→   102→      <input
   103→   103→        ref={inputRef}
   104→   104→        type="file"
   105→   105→        accept="image/jpeg,image/png,image/webp"
   106→   106→        className="hidden"
   107→   107→        onChange={(e) => {
   108→   108→          const file = e.target.files?.[0];
   109→   109→          if (file) handleFile(file);
   110→   110→        }}
   111→   111→      />
   112→   112→      <button
   113→   113→        onClick={() => inputRef.current?.click()}
   114→   114→        disabled={uploading}
   115→   115→        className="inline-flex items-center gap-2 rounded-lg border border-champagne/20 bg-ink/50 px-3 py-2 text-xs font-semibold text-champagne transition-colors hover:bg-champagne/10 disabled:opacity-40"
   116→   116→      >
   117→   117→        {uploading ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Upload className="h-3.5 w-3.5" />}
   118→   118→        {uploading ? "Uploading…" : label}
   119→   119→      </button>
   120→   120→    </div>
   121→   121→  );
   122→   122→}
